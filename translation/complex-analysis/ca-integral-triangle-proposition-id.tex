\begin{prop} \label{prop:inttriangleineq}
Misalkan $f \colon [a,b] \to \C$ terintegralkan (Riemann).
Maka $\sabs{f}$ terintegralkan (Riemann) dan
\begin{equation*}
\abs{\int_a^b f(t) \, dt} \leq 
\int_a^b \abs{f(t)} \, dt .
\end{equation*}
\end{prop}
